
Now here we had added global middle property.
this middleware will be called for every api call.
------------------------------------
app.use((req, res, next) => {
  console.log('middleware')
  console.log(req.url)
  next()
})

app.get('/',  (req, res) => {
  console.log('home ')
  console.log(req.url)
  res.send('Home Page');
})

app.get('/test', (req, res) => {
  console.log('test')
  console.log(req.url);
  console.log(req.route)
  res.send('Test Page');
})

// Note: In this case we will get req.url value same in 
// both middleware and APIs.
